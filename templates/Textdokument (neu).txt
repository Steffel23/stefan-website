        window.onload = function() {

            let upLvlLaser = 1; let upLvlShield = 1; let upLvlSpeed = 1;
            let upCostLaser = 20; let upCostShield = 35; let upCostSpeed = 15;

            const player = {
                x: 300, y: 280, width: 32, height: 32, speed: 4.5, maxShield: 100, shield: 100,
                movingLeft: false, movingRight: false, movingUp: false, movingDown: false
            };

            let stars = []; let lasers = []; let enemies = []; let crystals = []; let spawnTimer = 0;

            for(let i=0; i<40; i++) {
                stars.push({ x: Math.random()*600, y: Math.random()*380, size: Math.random()*1.5+0.5, speed: Math.random()*1+0.5 });
            }

            // PC-Steuerung
            document.addEventListener('keydown', (e) => {
                if (e.key === 'ArrowLeft' || e.key === 'a') player.movingLeft = true;
                if (e.key === 'ArrowRight' || e.key === 'd') player.movingRight = true;
                if (e.key === 'ArrowUp' || e.key === 'w') player.movingUp = true;
                if (e.key === 'ArrowDown' || e.key === 's') player.movingDown = true;
                if (e.key === ' ') { e.preventDefault(); fireLaser(); }
            });
            document.addEventListener('keyup', (e) => {
                if (e.key === 'ArrowLeft' || e.key === 'a') player.movingLeft = false;
                if (e.key === 'ArrowRight' || e.key === 'd') player.movingRight = false;
                if (e.key === 'ArrowUp' || e.key === 'w') player.movingUp = false;
                if (e.key === 'ArrowDown' || e.key === 's') player.movingDown = false;
            });

            // Handy-Steuerung
            function setupTouch(id, prop) {
                const btn = document.getElementById(id);
                btn.addEventListener('touchstart', (e) => { e.preventDefault(); player[prop] = true; });
                btn.addEventListener('touchend', () => player[prop] = false);
            }
            setupTouch('p-btn-left', 'movingLeft'); setupTouch('p-btn-right', 'movingRight');
            setupTouch('p-btn-up', 'movingUp'); setupTouch('p-btn-down', 'movingDown');
            document.getElementById('p-btn-fire').addEventListener('touchstart', (e) => { e.preventDefault(); fireLaser(); });

            // Upgrade-Zentrale Logik
            document.getElementById('up-laser').addEventListener('click', () => {
                if (credits >= upCostLaser) {
                    credits -= upCostLaser; upLvlLaser++; upCostLaser = Math.floor(upCostLaser * 1.6);
                    document.getElementById('lvl-laser').innerText = upLvlLaser; document.getElementById('cost-laser').innerText = upCostLaser + " 💎";
                    updateUI();
                }
            });
